## 问题 1: ESP8266 串口接收竞态条件 (严重)

### 问题验证 ✅ 确认存在

**位置:** esp8266.c 第 89-92 行, 407-416 行

**问题描述:**
- `rx_write_idx` 在中断回调中被修改
- `ESP8266_SendATCommand` 中无保护地清零索引
- 可能导致数据竞争和缓冲区损坏

**修复方案:**

```c
// esp8266.c - 修改 ESP8266_SendATCommand 函数
uint8_t ESP8266_SendATCommand(const char *cmd, const char *expected_resp, uint32_t timeout_ms)
{
    uint8_t cmd_buf[ESP8266_TX_BUF_SIZE];
    uint32_t start_tick;
    
    /* 进入临界区，防止中断修改缓冲区 */
    taskENTER_CRITICAL();
    // 或: HAL_NVIC_DisableIRQ(USART3_IRQn);
    
    /* 清空接收缓冲区 */
    rx_write_idx = 0;
    rx_read_idx = 0;
    memset(rx_buffer, 0, ESP8266_RX_BUF_SIZE);
    
    /* 退出临界区 */
    taskEXIT_CRITICAL();
    // 或: HAL_NVIC_EnableIRQ(USART3_IRQn);
    
    /* 组装命令 */
    snprintf((char *)cmd_buf, ESP8266_TX_BUF_SIZE, "%s\r\n", cmd);
    
    /* 发送命令 */
    HAL_UART_Transmit(&huart3, cmd_buf, strlen((char *)cmd_buf), 1000);
    
    /* 等待响应 */
    start_tick = HAL_GetTick();
    while ((HAL_GetTick() - start_tick) < timeout_ms) {
        /* 检查期望响应 */
        if (expected_resp != NULL) {
            if (strstr((char *)rx_buffer, expected_resp) != NULL) {
                return 0;
            }
        }
        
        /* 检查错误响应 */
        if (strstr((char *)rx_buffer, "ERROR") != NULL) {
            return 1;
        }
        
        HAL_Delay(10);
    }
    
    return 1;  // 超时
}
```

## 问题 2: MQTT任务信号量可能累积 (中等)

### 问题验证 ✅ 确认存在

**位置:** app_tasks.c 第 96-99 行

**问题描述:**
- SensorTask 每5秒释放信号量，不管MQTT是否处理完
- 如果MQTT任务阻塞，信号量会累积
- 可能导致MQTT任务连续发布多次

**修复方案:**

```c
// app_tasks.c - 修改 SensorTask 函数
void SensorTask(void *argument)
{
    uint32_t last_mqtt_tick = 0;
    
    printf("[SensorTask] Started\r\n");
    
    for (;;) {
        /* 读取所有传感器数据 */
        SensorManager_ReadAll();
        
        /* 检查是否需要触发MQTT发布 (每5秒) */
        /* 修复: 检查MQTT是否已连接，避免信号量累积 */
        if ((HAL_GetTick() - last_mqtt_tick) >= MQTT_PUBLISH_INTERVAL_MS) {
            /* 只在MQTT连接状态下释放信号量 */
            if (ESP8266_GetState() == ESP8266_STATE_MQTT_CONNECTED) {
                /* 尝试获取信号量，如果已存在则不再释放 */
                if (osSemaphoreAcquire(mqttPublishSem, 0) != osOK) {
                    /* 信号量不存在，释放一个 */
                    osSemaphoreRelease(mqttPublishSem);
                }
                // 如果信号量已存在，说明MQTT任务还未处理，不累积
            }
            last_mqtt_tick = HAL_GetTick();
        }
        
        /* 100ms采样间隔 */
        osDelay(SENSOR_SAMPLE_INTERVAL_MS);
    }
}
```

**替代方案 - 使用计数信号量:**

```c
// app_tasks.h - 修改信号量创建
// 创建计数信号量，最大计数为1，防止累积
#define mqttPublishSem osSemaphoreNew(1, 0, NULL)  // 保持原样，但使用时检查

// 或使用二进制信号量，确保只有一个发布请求
```

## 问题 3: printf 重入问题 (中等)

### 问题验证 ✅ 确认存在

**问题描述:**
- 多个任务调用 printf 输出到同一个串口
- 没有互斥保护，可能导致输出混乱
- 在嵌入式系统中，printf 通常不是线程安全的

**修复方案:**

### 方案 A: 添加互斥锁 (推荐)

```c
// debug_log.h - 添加互斥锁声明
#ifndef __DEBUG_LOG_H
#define __DEBUG_LOG_H

#include "main.h"
#include <stdio.h>

// ... 现有代码 ...

/* 添加互斥锁 */
extern osMutexId_t printfMutex;
void DebugLog_InitMutex(void);

// ... 现有代码 ...

#endif
```

```c
// debug_log.c - 实现互斥锁保护
#include "debug_log.h"
#include <stdarg.h>
#include <string.h>

extern UART_HandleTypeDef huart1;
osMutexId_t printfMutex = NULL;

void DebugLog_InitMutex(void)
{
    if (printfMutex == NULL) {
        printfMutex = osMutexNew(NULL);
    }
}

void DebugLog(LogLevel_t level, LogModule_t module, const char *fmt, ...)
{
    if (level < g_current_log_level) {
        return;
    }
    
    char buffer[256];
    int len = 0;
    
    /* 获取互斥锁 */
    if (printfMutex != NULL) {
        osMutexAcquire(printfMutex, osWaitForever);
    }
    
    /* 添加时间戳 */
    #if ENABLE_LOG_TIMESTAMP
    len += snprintf(buffer + len, sizeof(buffer) - len, "[%lu]", HAL_GetTick());
    #endif
    
    /* 添加日志级别 */
    len += snprintf(buffer + len, sizeof(buffer) - len, "[%s]", DebugLog_GetLevelString(level));
    
    /* 添加模块 */
    #if ENABLE_LOG_MODULE
    len += snprintf(buffer + len, sizeof(buffer) - len, "[%s]", DebugLog_GetModuleString(module));
    #endif
    
    /* 添加分隔符 */
    len += snprintf(buffer + len, sizeof(buffer) - len, " ");
    
    /* 添加内容 */
    va_list args;
    va_start(args, fmt);
    int msg_len = vsnprintf(buffer + len, sizeof(buffer) - len - 3, fmt, args);
    va_end(args);
    
    if (msg_len >= (int)(sizeof(buffer) - len - 3)) {
        strcat(buffer, "...");
        len = strlen(buffer);
    } else {
        len += msg_len;
    }
    
    /* 添加换行 */
    len += snprintf(buffer + len, sizeof(buffer) - len, "\r\n");

    /* 输出到串口 */
    HAL_UART_Transmit(&huart1, (uint8_t*)buffer, len, 10);
    
    /* 释放互斥锁 */
    if (printfMutex != NULL) {
        osMutexRelease(printfMutex);
    }
}
```

### 方案 B: 使用安全的日志函数

```c

// 原代码:
```
